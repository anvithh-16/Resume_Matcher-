1"""Resume-job matching engine (FR-06), skill gap (FR-07) and explanations (FR-09). 

2 

3Owner: Anirudh N (PES1UG24AM041) 

4Implements SAD 4.2 exactly: 

5 score = 100 x (0.75 x skill_overlap + 0.25 x keyword_similarity), clamped to [0, 100] 

6 skill_overlap = matched_required_skills / number_of_required_skills 

7 keyword_similarity = cosine similarity of resume text and job title + description 

8Fallback when a job has no required skills: score = 100 x keyword_similarity. 

9""" 

10import math 

11import re 

12from collections import Counter 

13 

14from app.services.skills import normalize_skill 

15 

16SKILL_WEIGHT, KEYWORD_WEIGHT = 0.75, 0.25 

17TOKEN = re.compile(r"[a-z0-9+#.]+") 

18STOPWORDS = {"and", "the", "a", "an", "of", "to", "in", "for", "with", "on", "is", "are", "we", "you", 

19 "our", "your", "using", "role", "work", "will", "be", "as", "at", "by", "or"} 

20 

21 

22def _vector(text: str) -> Counter: 

23 return Counter(t for t in TOKEN.findall((text or "").lower()) if t not in STOPWORDS) 

24 

25 

26def cosine_similarity(a: str, b: str) -> float: 

27 va, vb = _vector(a), _vector(b) 

28 dot = sum(va[t] * vb[t] for t in va.keys() & vb.keys()) 

29 norm = math.sqrt(sum(v * v for v in va.values())) * math.sqrt(sum(v * v for v in vb.values())) 

30 return 0.0 if norm == 0 else dot / norm 

31 

32 

33def _join(items: list[str]) -> str: 

34 return items[0] if len(items) == 1 else ", ".join(items[:-1]) + " and " + items[-1] 

35 

36 

37def build_explanation(matched: list[str], missing: list[str], has_required: bool) -> list[str]: 

38 """FR-09: every sentence is backed by the job's requirements or the parsed resume.""" 

39 lines = [] 

40 if matched: 

41 lines.append(f"Your resume lists {_join(matched)}, which this job requires.") 

42 if missing: 

43 verb = "is" if len(missing) == 1 else "are" 

44 lines.append(f"{_join(missing)} {verb} required but not found in your parsed resume.") 

45 if not has_required: 

46 lines.append("This job lists no required skills, so the score uses text similarity only.") 

47 return lines 

48

50def compute_match(profile: dict, job) -> dict: 

51 resume_skills = {normalize_skill(s) for s in (profile.get("skills") or [])} 

52 required = list(dict.fromkeys(normalize_skill(s) for s in (job.required_skills or []))) 

53 matched = [s for s in required if s in resume_skills] 

54 missing = [s for s in required if s not in resume_skills] 

55 keyword = cosine_similarity(profile.get("raw_text", ""), f"{job.title} {job.description}") 

56 if required: 

57 overlap = len(matched) / len(required) 

58 score = 100 * (SKILL_WEIGHT * overlap + KEYWORD_WEIGHT * keyword) 

59 else: 

60 overlap = 0.0 

61 score = 100 * keyword # documented fallback, no division by zero 

62 return { 

63 "job_id": job.id, 

64 "score": round(max(0.0, min(100.0, score)), 1), 

65 "matched_skills": matched, 

66 "missing_skills": missing, 

67 "components": {"skill_overlap": round(overlap, 2), "keyword_similarity": round(keyword, 2)}, 

68 "explanation": build_explanation(matched, missing, bool(required)), 

69 }
