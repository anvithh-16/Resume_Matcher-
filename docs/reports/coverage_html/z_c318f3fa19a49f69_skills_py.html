1"""Skill normalization and extraction (FR-07). 

2 

3Owner: Anirudh N (PES1UG24AM041) 

4Ankitha's job data and parser use ONLY the canonical names (the keys of SKILL_ALIASES). 

5""" 

6import re 

7 

8# Canonical skill -> aliases that mean the same thing. 

9SKILL_ALIASES: dict[str, list[str]] = { 

10 "python": ["python3"], 

11 "java": [], 

12 "javascript": ["ecmascript"], 

13 "typescript": [], 

14 "c++": ["cpp"], 

15 "sql": ["mysql", "postgresql", "postgres", "sqlite", "t-sql"], 

16 "fastapi": ["fast api"], 

17 "django": [], 

18 "flask": [], 

19 "react": ["reactjs", "react.js"], 

20 "node.js": ["nodejs", "node"], 

21 "html": ["html5"], 

22 "css": ["css3"], 

23 "git": ["github", "gitlab"], 

24 "docker": [], 

25 "kubernetes": ["k8s"], 

26 "aws": ["amazon web services"], 

27 "azure": [], 

28 "linux": [], 

29 "machine learning": ["ml"], 

30 "deep learning": [], 

31 "nlp": ["natural language processing"], 

32 "pandas": [], 

33 "numpy": [], 

34 "tensorflow": [], 

35 "pytorch": ["torch"], 

36 "scikit-learn": ["sklearn", "scikit learn"], 

37 "data analysis": ["data analytics"], 

38 "rest api": ["rest apis", "restful", "restful api"],

40 "excel": ["ms excel", "microsoft excel"], 

41 "tableau": [],

44 "communication": ["communication skills"], 

45} 

46 

47_ALIAS_TO_CANON = {term: canon for canon, aliases in SKILL_ALIASES.items() for term in [canon, *aliases]} 

48 

49# Longest terms first. A term only matches as a whole word, so "java" never matches "javascript" 

50# and "excel" never matches "excellent". 

51_PATTERNS = [(re.compile(rf"(?<![a-z0-9]){re.escape(t)}(?![a-z0-9])"), _ALIAS_TO_CANON[t]) 

52 for t in sorted(_ALIAS_TO_CANON, key=len, reverse=True)] 

53 

54 

55def normalize_skill(name: str) -> str: 

56 """' Python3 ' -> 'python'; unknown skills are returned lower-cased.""" 

57 cleaned = re.sub(r"\s+", " ", (name or "").strip().lower()) 

58 return _ALIAS_TO_CANON.get(cleaned, cleaned) 

59 

60 

61def extract_skills(text: str) -> list[str]: 

62 """Return the sorted canonical skills found in free text (never invents skills).""" 

63 low = re.sub(r"\s+", " ", (text or "").lower()) 

64 return sorted({canon for pattern, canon in _PATTERNS if pattern.search(low)})
